package dev.notai.tracker

import java.io.File
import org.junit.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class LegacyMatrixTest {
    @Test fun compareSameExportBytesBeforePreview() {
        val dir = File(System.getenv("TRACKER_MATRIX_DIR"))
        val label = System.getenv("TRACKER_MATRIX_LABEL")
        val fixed = System.getenv("TRACKER_MATRIX_FIXED") == "true"
        val output = StringBuilder()
        for(name in listOf("original-v1.json","app-v15.json")) {
            val target = Tracker(MemoryStore(),FakeClock(wall = 3000))
            val before = target.data();val backup = Backups(target,MemoryRollback())
            val file = File(dir,name)
            if(name == "original-v1.json" && !fixed) {
                val error = assertFailsWith<CoreError> { backup.prepare { file.inputStream() } }
                assertEquals(ErrorCode.UNSUPPORTED_IMPORT,error.code)
                assertEquals("Missing dataset fields",error.message)
                output.append("$label $name: rejected before preview: ${error.message}\n")
            } else {
                val preview = backup.prepare { file.inputStream() }
                assertEquals("matrix-data",preview.datasetId)
                assertEquals(1,preview.cards);assertEquals(1,preview.sessions);assertEquals(1,preview.segments)
                assertEquals("Europe/Moscow",preview.settings.reportZone)
                assertEquals(before,target.data())
                backup.apply(preview)
                val restored = target.data()
                assertEquals("matrix-card",restored.cards.single().id)
                assertEquals("matrix-favorite",restored.favorites.single().id)
                assertEquals("matrix-session",restored.sessions.single().id)
                assertEquals("matrix-segment",restored.segments.single().id)
                assertEquals("matrix-measurement",restored.measurements.single().id)
                if(name == "app-v15.json") assertEquals(false,restored.settings.surfaces.shortcut)
                output.append("$label $name: preview and restore passed\n")
            }
            if(name == "original-v1.json" && !fixed) assertEquals(before,target.data())
        }
        File(dir,"$label.txt").writeText(output.toString())
    }
}
